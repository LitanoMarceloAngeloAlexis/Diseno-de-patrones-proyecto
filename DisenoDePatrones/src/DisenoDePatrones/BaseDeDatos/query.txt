-- Procedimiento almacenado para SELECT
CREATE PROCEDURE sp_SelectFromTable
    @tableName NVARCHAR(50)
AS
BEGIN
    DECLARE @sql NVARCHAR(MAX);
    SET @sql = 'SELECT * FROM ' + QUOTENAME(@tableName);
    EXEC sp_executesql @sql;
END;

-- Procedimiento almacenado para ALTER
CREATE PROCEDURE sp_AlterRecord
    @tableName NVARCHAR(50),
    @DNI VARCHAR(20),
    @columna NVARCHAR(50),
    @nuevoValor NVARCHAR(100)
AS
BEGIN
    DECLARE @sql NVARCHAR(MAX);
    SET @sql = 'UPDATE ' + QUOTENAME(@tableName) + ' SET ' + QUOTENAME(@columna) + ' = @nuevoValor WHERE DNI = @DNI';
    EXEC sp_executesql @sql, N'@DNI VARCHAR(20), @nuevoValor NVARCHAR(100)', @DNI, @nuevoValor;
END;

CREATE PROCEDURE sp_GetCiudadanosComunes
AS
BEGIN
    SELECT *
    FROM CIUDADANO
    WHERE DNI NOT IN (
        SELECT DNI FROM AGENTEPUBLICO
        UNION
        SELECT DNI FROM FUERZAORDEN
    );
END;

CREATE PROCEDURE sp_GetAgentesPublicos
AS
BEGIN
    SELECT 
        c.DNI,
        c.NOMBRE,
        c.APELLIDO,
        c.EDAD,
        c.PROCEDENCIA,
        a.CARGO
    FROM CIUDADANO c
    INNER JOIN AGENTEPUBLICO a ON c.DNI = a.DNI;
END;

CREATE PROCEDURE sp_GetFuerzaOrden
AS
BEGIN
    SELECT 
        c.DNI,
        c.NOMBRE,
        c.APELLIDO,
        c.EDAD,
        c.PROCEDENCIA,
        f.RANGO,
        f.CARGO
    FROM CIUDADANO c
    INNER JOIN FUERZAORDEN f ON c.DNI = f.DNI;
END;

CREATE PROCEDURE autenticar_login
    @dni VARCHAR(9),
    @contraseña_ingresada NVARCHAR(100),
    @autenticado BIT OUTPUT
AS
BEGIN
    DECLARE @contraseña_guardada NVARCHAR(100);

    SELECT @contraseña_guardada = contraseña
    FROM login_ciudadano
    WHERE dni = @dni;

    IF @contraseña_guardada IS NOT NULL AND @contraseña_guardada = @contraseña_ingresada
        SET @autenticado = 1;
    ELSE
        SET @autenticado = 0;
END;

CREATE PROCEDURE sp_InsertarTramite
    @dni VARCHAR(20),
    @titulo VARCHAR(255)
AS
BEGIN
    SET NOCOUNT ON;

    IF EXISTS (
        SELECT 1
        FROM Tramites
        WHERE DNI = @dni
          AND Titulo = @titulo
          AND Estado IN ('Pendiente', 'En proceso')
    )
    BEGIN
        RAISERROR('Ya existe un trámite activo de este tipo para el ciudadano.', 16, 1);
        RETURN;
    END

    INSERT INTO Tramites (DNI, Titulo, Estado)
    VALUES (@dni, @titulo, 'En proceso');
END;

CREATE PROCEDURE sp_GetTramitesPorDNI
    @dni NVARCHAR(20)
AS
BEGIN
    SET NOCOUNT ON;

    SELECT ID, DNI, Titulo, Estado
    FROM Tramites
    WHERE DNI = @dni AND Estado IN ('Pendiente', 'En proceso')
END

CREATE PROCEDURE sp_GetTramitesPendientes
AS
BEGIN
    SELECT 
        Id,
        Dni,
        Titulo,
        Estado
    FROM Tramites
    WHERE Estado IN ('Pendiente', 'En Proceso')
    ORDER BY Id DESC;
END

CREATE TABLE Documentos (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    Dni NVARCHAR(20) NOT NULL,
    NombreArchivo NVARCHAR(255) NOT NULL,
    TipoMime NVARCHAR(100),
    Tamaño INT,
    FechaSubida DATETIME DEFAULT GETDATE(),
    Archivo VARBINARY(MAX)
);

CREATE PROCEDURE sp_InsertarDocumento
    @Dni NVARCHAR(20),
    @NombreArchivo NVARCHAR(255),
    @TipoMime NVARCHAR(100),
    @Tamaño INT,
    @Archivo VARBINARY(MAX)
AS
BEGIN
    INSERT INTO Documentos (Dni, NombreArchivo, TipoMime, Tamaño, Archivo)
    VALUES (@Dni, @NombreArchivo, @TipoMime, @Tamaño, @Archivo);
END

CREATE PROCEDURE sp_ObtenerDocumentosPorDNI
    @Dni NVARCHAR(20)
AS
BEGIN
    SELECT 
        Id,
        NombreArchivo,
        TipoMime,
        Tamaño,
        FechaSubida
    FROM Documentos
    WHERE Dni = @Dni
    ORDER BY FechaSubida DESC;
END

ALTER PROCEDURE sp_ObtenerDocumentoPorId
    @Id INT
AS
BEGIN
    SELECT 
        Id,
        NombreArchivo,
        TipoMime,
        Archivo
    FROM Documentos
    WHERE Id = @Id;
END

CREATE PROCEDURE sp_EliminarDocumentoPorId
    @Id INT
AS
BEGIN
    DELETE FROM Documentos WHERE Id = @Id;
END
